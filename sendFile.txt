use std::fs::File;
use std::io::{Read, Write};
use std::net::TcpListener;
pub fn start_server() {
    let listener = TcpListener::bind("0.0.0.0:9000").expect("Failed to bind");

    println!("Listening on 9000");

    for stream in listener.incoming() {
        match stream {
            Ok(mut stream) => {
                // Read message length (4 bytes)
                let mut length_buffer = [0u8; 2];

                stream.read_exact(&mut length_buffer).unwrap();

                let file_name_length = u16::from_be_bytes(length_buffer);

                println!("Length = {}", file_name_length);

                // FileName

                let mut file_name_buffer = vec![0u8; file_name_length as usize];

                stream.read_exact(&mut file_name_buffer).unwrap();
                let file_name = String::from_utf8(file_name_buffer).unwrap();
                println!("{:?}", file_name);

                //FileSize
                let mut file_size_buffer = [0u8; 8];
                stream.read_exact(&mut file_size_buffer).unwrap();
                let file_size = u64::from_be_bytes(file_size_buffer);

                // Create File
                let mut file = File::create(file_name+"new").unwrap();
                let mut received = 0u64;
                let mut buffer = [0u8; 8192];
                while received < file_size {
                    println!("ass");
                    let size = stream.read(&mut buffer).unwrap();
                    if size == 0 {
                        break;
                    }
                    file.write_all(&buffer[..size]).unwrap();
                    received += size as u64;
                    println!("Received {}/{} bytes", received, file_size);
                }
            }

            Err(e) => {
                println!("{}", e);
            }
        }
    }
}
